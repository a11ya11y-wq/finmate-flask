151 def _calculate_start_date(period): 

152 now = datetime.now() 

153 today_midnight = now.replace(hour=0, minute=0, second=0, microsecond=0) 

154 if period == 'week': 

155 start_date = today_midnight - timedelta(weeks=1) 

156 elif period == 'month':
